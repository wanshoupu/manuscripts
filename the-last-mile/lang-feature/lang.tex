\subsection{Packing and Unpacking}\label{subsec:packingAndUnpacking}

\begin{lstlisting}[language=Python]
    a,b,c=3,4,5
    print a,b,c # output: 3 4 5
    b,c,a=a,b,c
    print a,b,c # output: 5 3 4
\end{lstlisting}
